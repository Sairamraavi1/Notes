set timing on
spool VBAK_VBELN.txt

SELECT DISTINCT a.VBELN
FROM PSERP_R.VBAK a
JOIN PSERP_R.VBAP b
  ON a.VBELN = b.VBELN
WHERE a.ZZACC_TYPE = 'G'
  AND a.ZZACC_SUBTYPE = 'F'
  AND b.WERKS = 'DC20'
  AND a.ERDAT >= TO_DATE('2025-03-17','YYYY-MM-DD')
  AND a.ERDAT <  TO_DATE('2026-03-18','YYYY-MM-DD');

spool off;
exit
